155: /**
156:  * Percentage of the class characteristic cube strength, rounded to a whole
157:  * number. The reference prints 25.1 MPa as 84% and 21.5 MPa as 72% for a C30
158:  * mix, so the divisor is the class target — not the mean of the cubes.
159:  */
160: export const percentOfClass = (strength: number | null, classTarget: number | null): string => {
161:   if (strength === null || classTarget === null || classTarget <= 0) return "";
162:   return `${Math.round((strength / classTarget) * 100)}%`;
163: };

165: /**
166:  * Whether a cube is acceptable, as a fraction of the class characteristic
167:  * strength at the age it was broken.
168:  *
169:  * A bare comparison against the class figure would condemn every early cube:
170:  * the reference's own 7-day cube reached 84% of a C30 class and was reported
171:  * "Satisfactory". Each age is judged against the strength-gain table (1d 16%,
172:  * 3d 40%, 7d 65%, 14d 90%, 28d 99%), so a 28-day cube at 70% is below its
173:  * 99% expectation while a 3-day cube at 45% beats its 40% one. Past 28 days
174:  * the concrete is judged against the 28-day line, so a 55-day cube at 78%
175:  * reads "Unsatisfactory". Without a
176:  * known age the long-standing flat 65% line is kept as the fallback.
177:  */
178: export const isSatisfactory = (
179:   strength: number | null,
180:   classTarget: number | null,
181:   ageDays: number | null = null,
182: ): boolean => {
183:   if (strength === null || classTarget === null || classTarget <= 0) return false;
184:   // The strength-gain table stops at 28 days; older cubes hold the 28-day line.
185:   const cappedAge = ageDays === null ? null : Math.min(ageDays, 28);
186:   const expected = cappedAge === null ? 65 : expectedPercentAtAge(cappedAge) ?? 65;
187:   return strength >= (classTarget * expected) / 100;
188: };
