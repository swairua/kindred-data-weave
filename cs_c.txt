166:  * Whether a cube is acceptable, as a fraction of the class characteristic

169:  * A bare comparison against the class figure would condemn every early cube:
170:  * the reference's own 7-day cube reached 84% of a C30 class and was reported
171:  * "Satisfactory". Each age is judged against the strength-gain table (1d 16%,
172:  * 3d 40%, 7d 65%, 14d 90%, 28d 99%), so a 28-day cube at 70% is below its
173:  * 99% expectation while a 3-day cube at 45% beats its 40% one. Without a
174:  * known age the long-standing flat 65% line is kept as the fallback.
175:  */
176: export const isSatisfactory = (
177:   strength: number | null,
178:   classTarget: number | null,
179:   ageDays: number | null = null,
180: ): boolean => {
181:   if (strength === null || classTarget === null || classTarget <= 0) return false;
182:   const expected = ageDays === null ? 65 : expectedPercentAtAge(ageDays) ?? 65;
183:   return strength >= (classTarget * expected) / 100;
