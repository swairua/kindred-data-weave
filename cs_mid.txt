212: const drawCellText = (
213:   d: jsPDF,
214:   value: string,
215:   x: number,
216:   y: number,
217:   w: number,
218:   h: number,
219:   size: number,
220:   bold = false,
221: ) => {
222:   // Labels like "DATE OF\nCAST" and two-line "CM/5552\nCM/5553" lab refs sit on
223:   // a vertical stack of centred lines; a single line keeps the old placement.
224:   const lines = value
225:     .split("\n")
226:     .map((line) => line.trim())
227:     .filter((line) => line.length > 0);
228:   if (lines.length <= 1) {
229:     const fitted = fitText(d, value, w - 1.6, size);
230:     d.setFontSize(fitted);
231:     d.setFont("helvetica", bold ? "bold" : "normal");
232:     d.setTextColor(...COLORS.dark);
233:     // jsPDF anchors on the text baseline, so offset by a third of the cap height
234:     // to sit the text on the vertical centre of the cell.
235:     d.text(value, x + w / 2, y + h / 2 + fitted * 0.35, { align: "center" });
236:     return;
237:   }
238:   const longest = lines.reduce((a, b) => (a.length >= b.length ? a : b), "");
239:   const fitted = fitText(d, longest, w - 1.6, size);
240:   d.setFontSize(fitted);
241:   d.setFont("helvetica", bold ? "bold" : "normal");
242:   d.setTextColor(...COLORS.dark);
243:   const lineH = fitted * 0.42;
244:   let cy = y + h / 2 - ((lines.length - 1) * lineH) / 2;
245:   for (const line of lines) {
246:     d.text(line, x + w / 2, cy + fitted * 0.35, { align: "center" });
247:     cy += lineH;
248:   }
249: };
