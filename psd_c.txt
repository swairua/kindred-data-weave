128: /** Draw one row of ruled cells and return the y below it. Supports "\n" two-line captions. */
129: const drawCells = (d: jsPDF, x: number, y: number, h: number, cells: Cell[]): number => {
130:   let cx = x;
131:   d.setDrawColor(...COLORS.dark);
132:   d.setLineWidth(0.2);
133:   for (const cell of cells) {
134:     const shade = cell.label ? COLORS.labelBg : ([255, 255, 255] as [number, number, number]);
135:     d.setFillColor(shade[0], shade[1], shade[2]);
136:     d.rect(cx, y, cell.w, h, "FD");
137:     const value = (cell.text ?? "").trim();
138:     if (value) {
139:       const lines = value.split("\n").map((line) => line.trim()).filter(Boolean);
140:       const longest = lines.reduce((a, b) => (a.length >= b.length ? a : b), "");
141:       const size = fitText(d, longest, cell.w - 1.2, cell.size ?? 6.2, cell.minSize ?? 2.8);
142:       d.setFontSize(size);
143:       d.setFont("helvetica", cell.bold ? "bold" : "normal");
144:       d.setTextColor(...COLORS.dark);
145:       const align = cell.align ?? "left";
146:       const tx = align === "left" ? cx + 0.8 : align === "right" ? cx + cell.w - 0.8 : cx + cell.w / 2;
147:       const lineH = size * 0.38;
148:       const startY = lines.length > 1 ? y + h / 2 - lineH / 2 : y + h / 2;
149:       lines.forEach((line, index) => {
150:         d.text(line, tx, startY + index * lineH, { align, baseline: "middle" });
151:       });
152:     }
153:     cx += cell.w;
154:   }
155:   return y + h;
156: };
